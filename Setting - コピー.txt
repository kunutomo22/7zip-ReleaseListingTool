$LogLevel = "WARNING"   #この設定値以上のログが出力されます。
						# ERROR: エラーログ(エラーログのみ出力)
						# WARNING: 警告ログ(警告ログとエラーログのみ出力)
						# INFO: 情報ログ以上を出力する(デバッグログ以外全て出力)
						# None: 全てのログを出力する(デバッグログも出力)

$Popup = $false         #ポップアップを表示するかどうかの設定です。$trueにするとポップアップが表示されます。※一部のユーザの応答が必要なポップアップはこの設定に関わらず表示されます。

$OutType = "File"       #リストの出力形式を設定します。
						# File: ファイルに出力
						# Popup: ポップアップに出力 ※$Popup = $falseとするとリスト出力時のみポップアップが表示されます。
						# Console: コンソールに出力

$FileExtension = "csv"  #出力するファイルの拡張子を設定します。
						# csv: CSV形式で出力     ※1デフォルト
						#                        ※2想定外の拡張子を設定した場合はファイル名を想定外の拡張子にしCSV形式で出力されます。
						#
						# txt: TXT形式で出力     ※リリース履歴そのまま出力します。
						#
						# json: JSON形式で出力
						# md: Markdown形式で出力
